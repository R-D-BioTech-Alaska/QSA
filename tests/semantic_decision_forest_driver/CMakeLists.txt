cmake_minimum_required(VERSION 3.20)
project(QSASemanticDecisionForest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(semantic_decision_forest ../test_semantic_decision_forest.cpp)
add_executable(semantic_decision_forest_scale ../test_semantic_decision_forest_scale.cpp)
target_include_directories(semantic_decision_forest PRIVATE ../../include)
target_include_directories(semantic_decision_forest_scale PRIVATE ../../include)
if(MSVC)
    target_compile_options(semantic_decision_forest PRIVATE /W4)
    target_compile_options(semantic_decision_forest_scale PRIVATE /W4)
else()
    target_compile_options(semantic_decision_forest PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    target_compile_options(semantic_decision_forest_scale PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
