cmake_minimum_required(VERSION 3.20)
project(QSASemanticTripair LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(QSTATE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(QSTATE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(QSTATE_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(QSTATE_BUILD_PYTHON_TESTS OFF CACHE BOOL "" FORCE)
set(QSTATE_INSTALL_DEVELOPMENT OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. qsa-core)
add_executable(semantic_tripair_tests ../test_semantic_tripair.cpp)
target_link_libraries(semantic_tripair_tests PRIVATE qstate_core)
add_executable(semantic_tripair_benchmark ../../benchmarks/semantic_tripair.cpp)
target_link_libraries(semantic_tripair_benchmark PRIVATE qstate_core)
if(MSVC)
    target_compile_options(semantic_tripair_tests PRIVATE /W4)
else()
    target_compile_options(semantic_tripair_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
